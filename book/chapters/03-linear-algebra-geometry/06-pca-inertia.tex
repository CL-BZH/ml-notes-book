\chapter{PCA: Pairwise Distances and Total Inertia}\label{ch:pca-inertia}

Let's show that the average of all of pairs of square distances between the n 
individuals is equal to twice the total inertia.
\\

The average of all of pairs of square distances between the n individuals 
is equal to 
\begin{equation}
    \sum_{i=1}^{n} \sum_{j=1}^{n} p_{i}p_{j} (\textbf{e}_i - \textbf{e}_j)^T \textbf{M} (\textbf{e}_i - \textbf{e}_j) \label{pca-inertia:avg_d}
\end{equation}
Let's develop $(\textbf{e}_i - \textbf{e}_j)^T \textbf{M} (\textbf{e}_i - \textbf{e}_j)$.
\begin{align*}
    (\textbf{e}_i - \textbf{e}_j)^T \textbf{M} (\textbf{e}_i - \textbf{e}_j) &= ((\textbf{e}_i - g + g - \textbf{e}_j)^T \textbf{M} (\textbf{e}_i - g + g - \textbf{e}_j)) \\
    &= ((\textbf{e}_i - \textbf{g})^T \textbf{M} (\textbf{e}_i - \textbf{g})) + ((\textbf{e}_i - \textbf{g})^T \textbf{M} (\textbf{g} - \textbf{e}_j)) + 
    ((\textbf{g} - \textbf{e}_j)^T \textbf{M} (\textbf{e}_i - g )) + ((\textbf{g} - \textbf{e}_j)^T \textbf{M} (\textbf{g} - \textbf{e}_j)) \\
    &= d_{M}^2(\textbf{e}_i,\textbf{g}) + d_{M}^2(\textbf{e}_j,\textbf{g}) + 2 \times ((\textbf{e}_i - \textbf{g})^T \textbf{M} (\textbf{g} - \textbf{e}_j))
\end{align*}
Replacing in $\eqref{pca-inertia:avg_d}$ we get:
\begin{align*}
    \sum_{i=1}^{n} \sum_{j=1}^{n} p_{i}p_{j} (\textbf{e}_i - \textbf{e}_j)^T \textbf{M} (\textbf{e}_i - \textbf{e}_j) &=
    \sum_{i=1}^{n} \sum_{j=1}^{n} p_{i}p_{j} d_{M}^2(\textbf{e}_i,\textbf{g}) + 
    \sum_{i=1}^{n} \sum_{j=1}^{n} p_{i}p_{j} d_{M}^2(\textbf{e}_j,\textbf{g}) + 
    2 \sum_{i=1}^{n} \sum_{j=1}^{n} p_{i}p_{j} ((\textbf{e}_i - \textbf{g})^T \textbf{M} (\textbf{g} - \textbf{e}_j)) \\
    &= \underbrace{\sum_{j=1}^{n} p_{j}}_{= 1} \sum_{i=1}^{n} p_{i} d_{M}^2(\textbf{e}_i,\textbf{g}) + 
    \underbrace{\sum_{i=1}^{n} p_{i}}_{= 1} \sum_{j=1}^{n} p_{j} d_{M}^2(\textbf{e}_j,\textbf{g}) + 
    2 \sum_{i=1}^{n} \sum_{j=1}^{n} p_{i}p_{j} ((\textbf{e}_i - \textbf{g})^T \textbf{M} (\textbf{g} - \textbf{e}_j)) \\
    &= 2 \sum_{k=1}^{n} p_{k} d_{M}^2(\textbf{e}_k,\textbf{g}) + 
    2 \sum_{i=1}^{n} \sum_{j=1}^{n} p_{i}p_{j} ((\textbf{e}_i - \textbf{g})^T \textbf{M} (\textbf{g} - \textbf{e}_j)) \\
    &= 2 \sum_{k=1}^{n} p_{k} d_{M}^2(\textbf{e}_k,\textbf{g}) + 
    2 \sum_{i=1}^{n} p_{i} ((\textbf{e}_i - \textbf{g})^T \textbf{M} \sum_{j=1}^{n} p_{j} (\textbf{g} - \textbf{e}_j))\\
    &= 2 \sum_{k=1}^{n} p_{k} d_{M}^2(\textbf{e}_k,\textbf{g}) + 
    2 \sum_{i=1}^{n} p_{i} ((\textbf{e}_i - \textbf{g})^T \textbf{M} \big[ \textbf{g} - \underbrace{\sum_{j=1}^{n} p_{j} \textbf{e}_j}_{= g} \big]) \\
    &= 2 \sum_{k=1}^{n} p_{k} d_{M}^2(\textbf{e}_k,\textbf{g})
\end{align*}
\\

Now let's show that the total inertia is equal to the trace of $\textbf{M}\textbf{V}$.\\

Since $\textbf{M}$ is definite positive, it can be written as $M=\textbf{N}^T \textbf{N}$.\\

Thus, $p_{k} d_{M}^2(\textbf{e}_k,\textbf{g}) = p_{k} [\textbf{N}(\textbf{e}_k-\textbf{g})]^T [\textbf{N} (\textbf{e}_k - \textbf{g})] = p_k \textbf{y}_k^T \textbf{y}_k$,
where $\textbf{y}_k = \textbf{N}(\textbf{e}_k - \textbf{g})$.\\

Recall that  $\textbf{V} = \textbf{X}_c^T \textbf{D} \textbf{X}_c$ where

\begin{align*}
    \textbf{X}_c =
    \begin{bmatrix*}
        \horzbar &\dot{\textbf{e}_{1}}^T& \horzbar\\
        &\vdots&\\
        \horzbar &\dot{\textbf{e}_{i}}^T& \horzbar\\
        &\vdots&\\
        \horzbar &\dot{\textbf{e}_{n}}^T& \horzbar
    \end{bmatrix*} 
    =
    \begin{bmatrix*}
        \horzbar &[\textbf{e}_{1} - \textbf{g}]^T& \horzbar\\
        &\vdots&\\
        \horzbar &[\textbf{e}_{i} - \textbf{g}]^T& \horzbar\\
        &\vdots&\\
        \horzbar &[\textbf{e}_{n} - \textbf{g}]^T& \horzbar
    \end{bmatrix*}
\end{align*}

Therefore, $\textbf{M} \textbf{V} = \textbf{N}^T \textbf{N} \textbf{X}_c^T \textbf{D} \textbf{X}_c$, and 
$\Tr \big(\textbf{M} \textbf{V} \big) = \Tr \big(\textbf{N}^T \textbf{N} \textbf{X}_c^T \textbf{D} \textbf{X}_c \big)$.\\

Since $\Tr \big(\textbf{A} \textbf{B} \big) = \Tr \big(\textbf{B} \textbf{A} \big)$ we have 
$\Tr \big(\textbf{N}^T \textbf{N} \textbf{X}_c^T \textbf{D} \textbf{X}_c \big) = 
\Tr \big(\textbf{N} \textbf{X}_c^T \textbf{D} \textbf{X}_c \textbf{N}^T \big)$.\\

Computing $\textbf{N}\textbf{X}_c^T$, we get

\begin{align*}
    \textbf{N}\textbf{X}_c^T =
    \begin{bmatrix*}
        \vertbar & \cdots & \vertbar & \cdots & \vertbar\\
        \textbf{N}\dot{e}_{1} & \cdots & \textbf{N}\dot{e}_{j} & \cdots & \textbf{N}\dot{e}_{n}\\
        \vertbar & \cdots & \vertbar & \cdots & \vertbar
    \end{bmatrix*}
    =
    \begin{bmatrix*}
        \vertbar & \cdots & \vertbar & \cdots & \vertbar\\
        \dot{\textbf{y}}_{1} & \cdots & \dot{\textbf{y}}_{j} & \cdots & \dot{\textbf{y}}_{n}\\
        \vertbar & \cdots & \vertbar & \cdots & \vertbar
    \end{bmatrix*}
    = \textbf{Y}
\end{align*}

Thus, 
\begin{align*}
    \Tr\big(\textbf{M} \textbf{V} \big) = \Tr \big(\textbf{Y} \textbf{D} \textbf{Y}^T \big) &=
    \Tr \Bigg( \begin{bmatrix*}
        \vertbar & \cdots & \vertbar & \cdots & \vertbar\\
        {\textbf{y}}_{1} & \cdots & {\textbf{y}}_{j} & \cdots & {\textbf{y}}_{n}\\
        \vertbar & \cdots & \vertbar & \cdots & \vertbar
    \end{bmatrix*} 
    \textbf{D} 
    \begin{bmatrix*}
        \horzbar &\textbf{y}_{1}^T& \horzbar\\
        &\vdots&\\
        \horzbar &\textbf{y}_{i}^T& \horzbar\\
        &\vdots&\\
        \horzbar &\textbf{y}_{n}^T& \horzbar
    \end{bmatrix*}\Bigg) \\
    &= \Tr \Bigg( \begin{bmatrix*}
        \vertbar & \cdots & \vertbar & \cdots & \vertbar\\
        {\textbf{y}}_{1} & \cdots & {\textbf{y}}_{j} & \cdots & {\textbf{y}}_{n}\\
        \vertbar & \cdots & \vertbar & \cdots & \vertbar
    \end{bmatrix*} 
    \begin{bmatrix*}
        p_1(\horzbar &\textbf{y}_{1}^T& \horzbar)\\
        &\vdots&\\
        p_i(\horzbar &\textbf{y}_{i}^T& \horzbar)\\
        &\vdots&\\
        p_n(\horzbar &\textbf{y}_{n}^T& \horzbar)
    \end{bmatrix*}\Bigg)\\
    &= \Tr \Bigg(\sum_{k=1}^{n} 
    \begin{bmatrix*}
        \vertbar\\
        {\textbf{y}}_{k} \\
        \vertbar
    \end{bmatrix*}
    \begin{bmatrix*}
        p_k(\horzbar &\textbf{y}_{k}^T& \horzbar)
    \end{bmatrix*}\Bigg) \\
    &= \sum_{k=1}^{n} \Tr \Bigg(\begin{bmatrix*}
        \vertbar\\
        {\textbf{y}}_{k} \\
        \vertbar
    \end{bmatrix*}
    \begin{bmatrix*}
        p_k(\horzbar &\textbf{y}_{k}^T& \horzbar)
    \end{bmatrix*}\Bigg)  \\
    &= \sum_{k=1}^{n} \Tr \Bigg(
    \begin{bmatrix*}
        p_k(\horzbar &\textbf{y}_{k}^T& \horzbar)
    \end{bmatrix*}  
    \begin{bmatrix*}
        \vertbar\\
        {\textbf{y}}_{k} \\
        \vertbar
    \end{bmatrix*} \Bigg) \\
    &= \sum_{k=1}^{n} p_k \textbf{y}_k^T \textbf{y}_k = \sum_{k=1}^{n} p_{k} d_{M}^2(\textbf{e}_k,\textbf{g}) 
\end{align*}

\begin{definition}
    A matrix $\textbf{A} \in \mathcal{M}_{p\times p}(\mathbb{R})$ is $\textbf{M}$-symmetric if
    \begin{equation}
        \textbf{M}\textbf{A} = \textbf{A}^T \textbf{M}
    \end{equation}
\end{definition}

\begin{proposition}
A projector $\textbf{P}$ is an $\textbf{M}$-orthogonal projector iff $\textbf{P}$ is $\textbf{M}$-symmetric.
\end{proposition}

\begin{proof} Let $\textbf{P}$ be a projector ($\textbf{P}^2 = \textbf{P}$) on a subspace $\mathbb{E}_1$ of 
$\mathbb{E}$.\\

Then, $\forall{x, y} \in \mathbb{E}$ 
${\innerproduct{\textbf{P}\textbf{x}}{(\textbf{I} - \textbf{P})\textbf{y}}}_{\textbf{M}} = 0$.\\

That is, $\textbf{x}^T \textbf{P}^T \textbf{M} (\textbf{I} - \textbf{P})\textbf{y} =0$. \\

Or again, $\textbf{x}^T \textbf{P}^T \textbf{M} \textbf{y} = 
\textbf{x}^T \textbf{P}^T \textbf{M} \textbf{P} \textbf{y} $.

Taking for $\textbf{x}$ and $\textbf{y}$ the canonical basis vectors $x_k = 
\begin{bmatrix*}
    \vertbar\\
    {\textbf{1}}_{k} \\
    \vertbar
\end{bmatrix*}$ for $k = 1, \cdots, p$ 
and  
$y_l = 
\begin{bmatrix*}
    \vertbar\\
    {\textbf{1}}_{l} \\
    \vertbar
\end{bmatrix*}$ for $l = 1, \cdots, p$ 
we obtain that  $\textbf{P}^T \textbf{M} = \textbf{P}^T \textbf{M} \textbf{P}$.\\

Now, $\textbf{P}^T \textbf{M} \textbf{P}$ is symmetric (because $\textbf{M}$ is), thus $\textbf{P}^T \textbf{M}$ is symmetric,
i.e.\ $\textbf{P}^T \textbf{M} = (\textbf{P}^T \textbf{M})^T = \textbf{M} \textbf{P}$.\\

Conversely, if $\textbf{M}\textbf{P} = \textbf{P}^T\textbf{M}$, then
$\textbf{P}^T \textbf{M} (\textbf{I} - \textbf{P}) = \textbf{M}\textbf{P}(\textbf{I} - \textbf{P}) = \textbf{M}(\textbf{P} - \textbf{P}^2) = \textbf{0}$,
so $\innerproduct{\textbf{P}\textbf{x}}{(\textbf{I} - \textbf{P})\textbf{y}}_{\textbf{M}} = 0$ for all $\textbf{x}, \textbf{y}$: the projector is $\textbf{M}$-orthogonal.

\end{proof}

\begin{proposition}
    The $\textbf{M}$-orthogonal projector $\textbf{P}_{\textbf{a}}$ on the subspace $\operatorname{span}(\textbf{a})$ is 
    $P_{\textbf{a}} = \textbf{a} \big( \textbf{a}^T \textbf{M} \textbf{a}\big)^{-1} \textbf{a}^T \textbf{M}$.
\end{proposition}

\begin{proof} We have
    ${\innerproduct{\textbf{a}}{(\textbf{I} - \textbf{P}_{\textbf{a}}) \textbf{y}}}_{\textbf{M}} = 0$ \; 
    $\forall{\textbf{y}} \in \mathbb{E}$.\\

    That is, ${\textbf{a}}^T \textbf{M}  \textbf{y} = 
    {\textbf{a}}^T \textbf{M} \textbf{P}_{\textbf{a}} \textbf{y}$. \\

    Thus, writing $\textbf{P}_{\textbf{a}} \textbf{y} = \alpha \textbf{a}$ we obtain:\\

    $\alpha = \big( {\textbf{a}}^T \textbf{M} \textbf{a} \big)^{-1} 
    {\textbf{a}}^T \textbf{M}  \textbf{y}$. Hence, $\textbf{P}_{\textbf{a}} \textbf{y} = 
    \textbf{a} \alpha = 
    \textbf{a} \big( {\textbf{a}}^T \textbf{M} \textbf{a} \big)^{-1} {\textbf{a}}^T \textbf{M} \textbf{y}$.\\

    Thus, $\textbf{P}_{\textbf{a}} = \textbf{a} \big( {\textbf{a}}^T \textbf{M} \textbf{a} \big)^{-1} {\textbf{a}}^T \textbf{M}$.\\



\end{proof}
