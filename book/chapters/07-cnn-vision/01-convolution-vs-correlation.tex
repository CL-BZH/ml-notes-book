\chapter{Convolution versus Cross-Correlation}\label{ch:convolution-vs-correlation}

\section{Definitions}

Signals are indexed by integers and are taken to be zero outside their support.

\begin{enumerate}
\item \textbf{Convolution} flips the kernel before sliding it over the signal:
\[
(x * k)[n] = \sum_{m} x[m] \, k[n - m].
\]
The substitution $m' = n - m$ shows that it is commutative: \( x * k = k * x \).

\item \textbf{Cross-correlation} slides the kernel without flipping it:
\[
(x \star k)[n] = \sum_{m} x[m] \, k[m - n] = \sum_{m} x[m+n]\, k[m].
\]
The same substitution gives \( (k \star x)[n] = (x \star k)[-n] \): swapping the
arguments \emph{reverses} the output, so cross-correlation is \textbf{not}
commutative. Cross-correlation is a convolution with the flipped kernel
$\tilde{k}[m] = k[-m]$: \( x \star k = x * \tilde{k} \).
\end{enumerate}

\section{Example}

Take \( x = [x_1, x_2, x_3] \) (i.e.\ $x[1]=x_1$, $x[2]=x_2$, $x[3]=x_3$) and
\( k = [k_1, k_2] \), all other entries being $0$.

\subsection{Convolution}
\( (x * k)[n] = \sum_m x[m]\,k[n-m] \) is non-zero for $n = 2, \dots, 5$:
\[
\begin{array}{c|c}
n & (x * k)[n] = (k * x)[n] \\ \hline
2 & x_1 k_1 \\
3 & x_1 k_2 + x_2 k_1 \\
4 & x_2 k_2 + x_3 k_1 \\
5 & x_3 k_2
\end{array}
\]
Computing \( (k * x)[n] = \sum_m k[m]\,x[n-m] \) term by term gives exactly the same
values, e.g.\ \( (k * x)[3] = k_1 x_2 + k_2 x_1 \).

\subsection{Cross-correlation}
\( (x \star k)[n] = \sum_m x[m]\,k[m-n] \) is non-zero for $n = -1, \dots, 2$, while
\( (k \star x)[n] = \sum_m k[m]\,x[m-n] \) is non-zero for $n = -2, \dots, 1$:
\[
\begin{array}{c|c|c}
n & (x \star k)[n] & (k \star x)[n] \\ \hline
-2 & 0 & k_1 x_3 \\
-1 & x_1 k_2 & k_1 x_2 + k_2 x_3 \\
0 & x_1 k_1 + x_2 k_2 & k_1 x_1 + k_2 x_2 \\
1 & x_2 k_1 + x_3 k_2 & k_2 x_1 \\
2 & x_3 k_1 & 0
\end{array}
\]
For instance \( (x \star k)[1] = x[2]k[1] + x[3]k[2] = x_2k_1 + x_3k_2 \), whereas
\( (k \star x)[1] = k[2]x[1] = k_2 x_1 \). The two columns are mirror images of each other:
\( (k \star x)[n] = (x \star k)[-n] \).

\section{Key Observations}
\begin{enumerate}
\item \textbf{Convolution} is commutative: \( x * k = k * x \) (and also associative),
      which is why it is the natural operation in signal processing and probability
      (the density of a sum of independent random variables is the convolution of the densities).
\item \textbf{Cross-correlation} is not commutative: \( (k \star x)[n] = (x \star k)[-n] \).
      It measures the similarity between $x$ and shifted copies of $k$ (template matching).
\item The two operations differ only by flipping the kernel: \( x \star k = x * \tilde{k} \).
      In convolutional neural networks the kernel is learned, so flipping it is irrelevant:
      the ``convolution'' layers of deep-learning libraries actually compute a cross-correlation.
\end{enumerate}
