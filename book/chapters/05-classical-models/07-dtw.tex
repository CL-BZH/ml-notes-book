\begin{notechapter}
\chapter{Dynamic Time Warping}\label{ch:dtw}
\chaptersubtitle{Comparing sequences that run at different speeds}

\begin{chapterintro}
Two recordings of the same word, spoken at different speeds, are close as sounds but
far apart sample by sample: the Euclidean distance compares the $t$-th sample of one with
the $t$-th sample of the other, which may belong to different syllables. Dynamic time
warping (DTW) first aligns the two sequences, letting one of them wait while the other
advances, and then compares aligned samples. The best alignment is found by dynamic
programming in time proportional to the product of the lengths. You will learn
\begin{itemize}[nosep]
  \item alignment paths and the DTW recursion, on a worked example;
  \item a tested implementation, its cost, and the usual band constraint;
  \item why DTW is not a distance in the mathematical sense;
  \item soft-DTW, a differentiable relaxation, and what its gradient means.
\end{itemize}
\prerequisites{none beyond basic algorithms; \cref{sigmoid-softmax:sec:numerics} for the
log-sum-exp.}
\end{chapterintro}

\section{Alignments}\label{dtw:sec:alignment}

Let $x = (x_1,\dots,x_n)$ and $y = (y_1,\dots,y_m)$ be sequences of vectors in $\R^d$ (the lengths may differ),
and $c(a, b) = \norm{a - b}^q$ a local cost, usually with $q = 2$.

\begin{definition}[Alignment path]
An \emph{alignment path} is a sequence of index pairs $\pi = ((i_1, j_1),\dots,(i_K, j_K))$ such that
\begin{enumerate}[nosep]
  \item $(i_1, j_1) = (1, 1)$ and $(i_K, j_K) = (n, m)$ (the ends are matched);
  \item each step is $(1, 0)$, $(0, 1)$ or $(1, 1)$: $x$ advances, $y$ advances, or both.
\end{enumerate}
The \emph{DTW cost} is the smallest total cost over all paths, and the DTW distance its $q$-th root:
\[
  \mathrm{DTW}_q(x, y) = \Bigl(\min_\pi\sum_{(i,j)\in\pi}c(x_i, y_j)\Bigr)^{1/q} .
\]
\end{definition}

The conditions say that time runs forward in both sequences, no sample is skipped,
and a sample may be repeated (matched with several consecutive samples of the other
sequence) to absorb a difference of speed. The number of paths grows exponentially
with $n$ and $m$, but the cost is additive along the path, so the minimum satisfies a
recursion.

\begin{proposition}[DTW recursion]\label{dtw:prop:recursion}
Let $D(i, j)$ be the minimal cost of a path from $(1, 1)$ to $(i, j)$. With $D(0, 0) = 0$ and $D(i, 0) = D(0, j) = +\infty$
otherwise,
\[
  D(i, j) = c(x_i, y_j) + \min\bigl(D(i - 1, j),\ D(i, j - 1),\ D(i - 1, j - 1)\bigr),
\]
and $\mathrm{DTW}_q(x, y)^q = D(n, m)$.
\end{proposition}

\begin{proof}
The last step of an optimal path to $(i, j)$ comes from one of the three predecessors, and the
part of the path before it is optimal for that predecessor (otherwise replacing it would
lower the total). The infinite borders forbid paths that do not start at $(1, 1)$.
\end{proof}

The table $D$ is filled row by row in $O(nm)$ time; the optimal path is recovered by
walking back from $(n, m)$ to the predecessor with the smallest $D$ at each step.

\begin{example}[A worked table]\label{dtw:ex:table}
Let $x = (1, 3, 4, 2)$ and $y = (1, 2, 4, 3, 2)$ (scalars, $q = 2$). The local costs $(x_i - y_j)^2$ and the
cumulative costs $D(i, j)$ are
\[
  C = \begin{pmatrix}0&1&9&4&1\\4&1&1&0&1\\9&4&0&1&4\\1&0&4&1&0\end{pmatrix},\qquad
  D = \begin{pmatrix}\mathbf0&1&10&14&15\\4&\mathbf1&2&2&3\\13&5&\mathbf1&\mathbf2&6\\14&5&5&2&\mathbf2\end{pmatrix}.
\]
For instance $D(2, 3) = C(2, 3) + \min(D(1, 3), D(2, 2), D(1, 2)) = 1 + \min(10, 1, 1) = 2$. The cost is $D(4, 5) = 2$ and
$\mathrm{DTW}_2 = \sqrt2$. Walking back along the bold entries gives the path
$(1,1), (2,2), (3,3), (3,4), (4,5)$: $x_3 = 4$ is matched with both $y_3 = 4$ and $y_4 = 3$, and the total
cost $0 + 1 + 0 + 1 + 0 = 2$ comes from the pairs $(3, 2)$ and $(4, 3)$. The Euclidean distance is not even
defined here, since the lengths differ.
\end{example}

\section{Implementation and cost}\label{dtw:sec:implementation}

The following function implements \cref{dtw:prop:recursion} with an infinite border and
backtracking.

\begin{lstlisting}[language=Python]
import numpy as np

def dtw(x, y, q=2):
    """DTW between sequences of vectors x (n x d) and y (m x d).

    Returns (D[n, m] ** (1/q), optimal path as a list of (i, j), 0-based),
    where D is the cumulative cost with local cost ||x_i - y_j||^q."""
    x, y = np.atleast_2d(x), np.atleast_2d(y)
    n, m = len(x), len(y)
    D = np.full((n + 1, m + 1), np.inf)
    D[0, 0] = 0.0
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            cost = np.linalg.norm(x[i - 1] - y[j - 1]) ** q
            D[i, j] = cost + min(D[i - 1, j], D[i, j - 1], D[i - 1, j - 1])
    # backtrack from (n, m) to (1, 1)
    i, j, path = n, m, [(n - 1, m - 1)]
    while (i, j) != (1, 1):
        steps = [(i - 1, j - 1), (i - 1, j), (i, j - 1)]
        i, j = min(steps, key=lambda s: D[s])
        path.append((i - 1, j - 1))
    return D[n, m] ** (1.0 / q), path[::-1]

x = [[1], [2], [3], [4]]
y = [[1], [1], [2], [3], [4]]
dist, path = dtw(x, y)
print(dist, path)  # 0.0 [(0, 0), (0, 1), (1, 2), (2, 3), (3, 4)]
\end{lstlisting}

The two sequences of the example differ only by a repeated first sample, and DTW
matches $x_1$ with both copies: the distance is $0$.

\paragraph{Cost and constraints.} Time and memory are $O(nm)$; for long sequences this is
too much, and unconstrained paths may also produce pathological alignments (one
sample matched with half of the other sequence). The \emph{Sakoe--Chiba band} restricts
paths to $\abs{i - j}\leq w$ (for sequences of equal length), which reduces the cost to $O(nw)$ and
limits how far one sequence may run ahead; the \emph{Itakura parallelogram} limits the
local slope instead. Lower bounds such as LB\_Keogh, cheap to compute, let a nearest-
neighbour search discard most candidates without computing their DTW.

\begin{pitfall}[DTW is not a metric]
DTW is symmetric and nonnegative, but two different sequences can be at distance $0$ (a
sequence and its copy with a repeated sample), and the triangle inequality fails. With
$q = 1$ and absolute differences: $a = (0, 0)$, $b = (0, 1)$, $c = (0, 1, 1)$ give $\mathrm{DTW}(a, b) = 1$,
$\mathrm{DTW}(b, c) = 0$ and $\mathrm{DTW}(a, c) = 2>1 + 0$. Algorithms that rely on the triangle inequality
(metric trees, some clustering guarantees) do not apply as such.
\end{pitfall}

\section{Soft-DTW}\label{dtw:sec:soft}

The DTW cost is a minimum over paths, hence piecewise smooth and not differentiable
where two paths tie. To use it as a loss (to train a network to output sequences, or to
average sequences), Cuturi and Blondel (\emph{Soft-DTW: a Differentiable Loss Function for
Time-Series}, ICML 2017) replace the minimum by a smooth one.

\begin{definition}[Soft minimum, soft-DTW]
For $\gamma>0$,
\[
  \operatorname{min}^\gamma(a_1,\dots,a_k) = -\gamma\log\sum_{l=1}^ke^{-a_l/\gamma},
\]
and soft-DTW is obtained by using $\operatorname{min}^\gamma$ in the recursion of \cref{dtw:prop:recursion}.
\end{definition}

The soft minimum is $-\gamma\,\mathrm{LSE}(-a/\gamma)$, so by \cref{sigmoid-softmax:prop:lse}
\[
  \min_la_l - \gamma\log k\leq\operatorname{min}^\gamma(a_1,\dots,a_k)\leq\min_la_l,
\]
and it tends to the minimum as $\gamma\to0$; its gradient with respect to $a$ is the softmax
$\softmax(-a/\gamma)$. Unrolled, the soft-DTW recursion computes
\[
  \mathrm{sDTW}_\gamma(x, y) = -\gamma\log\sum_\pi\exp\Bigl(-\frac1\gamma\sum_{(i,j)\in\pi}c(x_i, y_j)\Bigr),
\]
a soft minimum over \emph{all} alignment paths, still in $O(nm)$ time. Its gradient with
respect to the cost matrix is the expected alignment matrix
$E_{ij} = P_\gamma\bigl((i, j)\in\pi\bigr)$ under the Gibbs distribution $P_\gamma(\pi)\propto\exp(-\mathrm{cost}(\pi)/\gamma)$ over paths, and a
backward recursion computes it in $O(nm)$ as well. As $\gamma\to0$ it concentrates on the optimal
path; for larger $\gamma$ it spreads over near-optimal ones.

\begin{pitfall}[Soft-DTW of a sequence with itself]
$\mathrm{sDTW}_\gamma(x, x)$ is not $0$; it can be negative, since every path, not only the diagonal,
contributes to the sum. Minimizing $\mathrm{sDTW}_\gamma(x, y)$ over $y$ therefore need not return $y = x$.
Blondel, Mensch and Vert (2021) proposed the \emph{soft-DTW divergence}
$\mathrm{sDTW}_\gamma(x, y) - \frac12\mathrm{sDTW}_\gamma(x, x) - \frac12\mathrm{sDTW}_\gamma(y, y)$ (\emph{Differentiable Divergences
Between Time Series}, AISTATS 2021). They prove it nonnegative and zero iff $x = y$ for
two modified costs (a log-augmented squared Euclidean cost, and the absolute value in
dimension~$1$); for the plain squared Euclidean cost they only conjecture it, with
numerical support.
\end{pitfall}

\paragraph{Further reading.} R.~Tavenard's blog posts \emph{An Introduction to Dynamic Time
Warping} and \emph{Differentiability of DTW and the Case of Soft-DTW} (2021) give
illustrated introductions; the \texttt{tslearn} library implements both, with
constraints and barycentres.

\begin{chaptersummary}
\begin{itemize}[nosep]
  \item An alignment path matches the ends and moves by $(1, 0)$, $(0, 1)$ or $(1, 1)$; DTW is the minimal
        total cost over paths, computed by $D(i, j) = c(x_i, y_j) + \min(D(i - 1, j), D(i, j - 1), D(i - 1, j - 1))$ in $O(nm)$.
  \item Bands and lower bounds make DTW practical on long sequences and large
        collections.
  \item DTW can vanish between different sequences and violates the triangle inequality.
  \item Soft-DTW replaces $\min$ by $-\gamma\log\sum e^{-a/\gamma}$; it is differentiable, and its gradient is the
        expected alignment under a Gibbs distribution over paths. Its divergence version is
        provably zero exactly on identical sequences for suitable costs (conjectured for the
        squared Euclidean one).
\end{itemize}
\end{chaptersummary}

Show that the number of alignment paths between sequences of lengths $n$ and $m$ is at least

\end{exercise}
